# 1. Установить зависимости
pip install -r requirements.txt

# 2. Создать БД в postgres
psql -U postgres

CREATE DATABASE coffee_db;
CREATE USER cafe_admin;
ALTER DATABASE coffee_db OWNER TO cafe_admin;
GRANT ALL PRIVILEGES ON DATABASE coffee_db TO cafe_admin;

GRANT pg_read_all_data TO cafe_admin;
GRANT pg_write_all_data TO cafe_admin;

# 3. Запустить сервер
uvicorn main:app --reload
